cmake_minimum_required(VERSION 3.16)
project(bridge_native_tests LANGUAGES CXX)
enable_testing()
add_executable(test_bridge test_bridge.cpp)
target_include_directories(test_bridge PRIVATE ../main)
target_compile_features(test_bridge PRIVATE cxx_std_17)
if(MSVC)
    target_compile_options(test_bridge PRIVATE /W4 /WX)
else()
    target_compile_options(test_bridge PRIVATE -Wall -Wextra -Wpedantic -Werror)
endif()
add_test(NAME bridge_interfaces COMMAND test_bridge)
